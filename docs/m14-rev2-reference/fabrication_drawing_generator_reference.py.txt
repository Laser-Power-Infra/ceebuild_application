def dim(ax,x1,y1,x2,y2,t,off=0,vert=False,fs=7):
    ax.annotate('',(x1,y1),(x2,y2),arrowprops=dict(arrowstyle='<->',lw=0.6))
    if vert: ax.text(x1+off,(y1+y2)/2,t,fontsize=fs,rotation=90,va='center',ha='center')
    else: ax.text((x1+x2)/2,y1+off,t,fontsize=fs,ha='center')
def panel(ax,x0,y0,W,typ,lab,s=0.12):
    # W panel width mm, height 2400; rails FL50x6 at bottom, middle, top; verticals 40x5
    H=2400
    X=lambda v:x0+v*s; Y=lambda v:y0+v*s
    for yc in (25,1200,2375):
        ax.add_patch(Rectangle((X(0),Y(yc-25)),W*s,50*s,fc='#bbbbbb',ec='k',lw=0.6))
        ax.plot([X(25)],[Y(yc)],'o',ms=2.5,mfc='w',mec='k'); ax.plot([X(W-25)],[Y(yc)],'o',ms=2.5,mfc='w',mec='k')
    if typ=='A':
        n=int(W/500)+1; xs=[min(max(i*500,20),W-20) for i in range(n)]
        for i in range(0,int(W/50)+1):
            ax.plot([X(i*50),X(i*50)],[Y(0),Y(H)],lw=0.15,color='#777')
        for j in range(0,int(H/50)+1):
            ax.plot([X(0),X(W)],[Y(j*50),Y(j*50)],lw=0.15,color='#777')
    else:
        n=int((W-40)//115)+1; xs=[ (W-(n-1)*115)/2 + i*115 for i in range(n)]
    for xv in xs:
        ax.add_patch(Rectangle((X(xv-20),Y(25)),40*s,2350*s,fc='#dddddd',ec='k',lw=0.4))
    dim(ax,X(0),Y(-120),X(W),Y(-120),f'{W}',off=-14)
    dim(ax,X(-120),Y(0),X(-120),Y(H),'2400',off=-12,vert=True)
    ax.text(X(0),Y(H)+18,lab,fontsize=9,weight='bold')
    return len(xs)
