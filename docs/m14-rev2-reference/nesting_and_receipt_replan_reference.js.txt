// first-fit decreasing on a fixed stock length
function ffd(lens,bin){const t=S.trim,k=S.kerf;lens=lens.slice().sort((a,b)=>b-a);const bins=[];
 for(const L of lens){let placed=false;for(const b of bins){const need=L+(b.c.length?k:0);if(b.r>=need){b.r-=need;b.c.push(L);placed=true;break;}}
  if(!placed){if(L>bin-t){bins.push({len:L+t,r:0,c:[L],over:1});}else bins.push({len:bin,r:bin-t-L,c:[L]});}}
 return bins;}
function patterns(bins){const m=new Map();for(const b of bins){const g={};b.c.forEach(L=>g[L]=(g[L]||0)+1);
 const key=b.len+'|'+Object.keys(g).sort((a,b)=>b-a).map(L=>L+'×'+g[L]).join(' + ');
 const o=m.get(key)||{len:b.len,pat:key.split('|')[1],n:0,used:b.len-S.trim-b.r,off:b.r};o.n++;m.set(key,o);}
 return [...m.values()].sort((a,b)=>b.n-a.n);}

function planIndent(){const out={rows:[],secs:{}};

  const g={};lines.forEach(p=>{(g[p.L]=g[p.L]||{L:p.L,q:0,ls:[]}).q+=p.q;g[p.L].ls.push(p);});
  for(const x of Object.values(g)){const k=S.kerf,usable=(sec==='ISMC 100X50'?S.mx100:S.custMax)-S.trim;let n=Math.floor((usable+k)/(x.L+k));
   const metres=x.L*x.q/1000;const forced=x.L>S.stdLen-S.trim;
   const nm=[...new Set(x.ls.map(p=>p.desc))];x.desc=nm[0]+(nm.length>1?` (+${nm.length-1} more)`:'');x.code=[...new Set(x.ls.map(p=>p.code).filter(Boolean))].join(', ');
   if((S.mode==='mixed'&&metres>=S.custThresh&&n>=2)||forced){if(n<1)n=1;let len=n*x.L+(n-1)*k+S.trim;

// re-plan on actually received bars
function replan(){const res={};RP={secs:res,frac:{}};
 for(const sec of SECS){const rec=(REC[sec]||[]).filter(r=>r.len>0&&r.n>0);
  const dem={};P.filter(p=>p.sec===sec).forEach(p=>dem[p.L]=(dem[p.L]||0)+p.q);
  const totalDem={...dem};const bars=[];rec.forEach(r=>{for(let i=0;i<r.n;i++)bars.push({len:r.len,r:r.len-S.trim,c:[]});});
  const k=S.kerf;
  // step 1: whole bars to exact multiples (customised lengths)
  const groups={};bars.forEach(b=>(groups[b.len]=groups[b.len]||[]).push(b));
  for(const [B,arr] of Object.entries(groups)){let free=arr.slice();
   const cands=Object.keys(dem).map(Number).map(L=>{const n=Math.floor((B-S.trim+k)/(L+k));return{L,n,w:B-S.trim-n*L-(n-1)*k};})
     .filter(c=>c.n>=2&&c.w<=S.tol+5).sort((a,b)=>a.w-b.w||b.L-a.L);
   for(const c of cands){if(!free.length)break;const take=Math.min(free.length,Math.floor(dem[c.L]/c.n));
    for(let i=0;i<take;i++){const b=free.pop();for(let j=0;j<c.n;j++)b.c.push(c.L);b.r=c.w;b.ex=1;}dem[c.L]-=take*c.n;}}
  // step 2: best-fit decreasing for the rest
  const left=[];Object.entries(dem).forEach(([L,q])=>{for(let i=0;i<q;i++)left.push(+L);});left.sort((a,b)=>b-a);
  const open=bars.filter(b=>!b.ex);const short=[];
  for(const L of left){let best=null;for(const b of open){const need=L+(b.c.length?k:0);if(b.r>=need&&(!best||b.r<best.r))best=b;}
   if(best){best.r-=L+(best.c.length?k:0);best.c.push(L);}else short.push(L);}
  const used=bars.filter(b=>b.c.length);const sup=ffd(short,S.stdLen);
  const sd={};short.forEach(L=>sd[L]=(sd[L]||0)+1);
  Object.keys(totalDem).forEach(L=>RP.frac[sec+'|'+L]=1-(sd[L]||0)/totalDem[L]);
  const offc={};used.forEach(b=>{if(b.r>=300){const k2=Math.floor(b.r/100)*100;offc[k2]=(offc[k2]||0)+1;}});
  res[sec]={recBars:bars.length,recMM:bars.reduce((s,b)=>s+b.len,0),used:used.length,idle:bars.length-used.length,
   pat:patterns(used),short:sd,shortN:short.length,sup,supByLen:sup.length,offc,need:Object.entries(totalDem).reduce((s,[L,q])=>s+L*q,0),
   cutMM:used.reduce((s,b)=>s+b.c.reduce((a,x)=>a+x,0),0)};}
}

function route(p){const t=typ(p.sec),d=p.desc.toUpperCase(),o={};const welded=/WELD|WELED/.test(d);
 o[t==='ch'||t==='an'?'cut1':'cut2']=1;
 if(t==='rd'){o.thread=1;o.bend=1;}
 else if(!welded){o.punch=t==='ch'?S.hCh:t==='an'?S.hAn:S.hFl;}
 const isClamp=d.includes('CLAMP');
 if(!isClamp&&/V[- ]CROSS ARM/.test(d)&&!d.includes('FLAT')){o.notch=1;o.bend=(o.bend||0)+1;o.weld=1;}
 if(isClamp)o.bend=(o.bend||0)+S.bends;
 if(!isClamp&&d.includes('BRACKET')&&p.L>=300)o.weld=1;
 if(!welded)o.num=1;return o;}
function prod(){const load={};OPS.forEach(o=>load[o.k]=0);const lines=[];

 return{rows,lines};}
